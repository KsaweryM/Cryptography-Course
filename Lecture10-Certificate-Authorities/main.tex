\begin{frame}{Why the CSR is signed}
  \vfill
  \begin{center}
    \begin{tikzpicture}
      \cert{c}{0,0}{sTeal}{3.6cm}{CSR}{\textbf{Subject:} bob.example\\\textbf{Public key:} 04 20 EF \ldots\\\textbf{Signature:} 30 45 \ldots}
      \node[hwmain, minimum width=2.2cm, minimum height=1.5cm] (v) at (5.6,-0.9) {\Large\faCogs\\[1pt]\small\bfseries verify};
      \draw[data] (c.east |- v.west) to node[below, font=\scriptsize, align=center] {the key and the\\signature from the CSR} (v.west);
      \node[font=\small, text=nInk, anchor=west, align=left] at (6.9,-0.9) {\ok\ True: whoever made it\\holds the private key};
    \end{tikzpicture}
  \end{center}
  \vfill
  \bigdef{\textbf{Proof of possession:} the request was made by whoever holds the private key.\\
    It does \textbf{not} prove that bob.example belongs to Bob.}
  \vfill
  \begin{center}\small Without it, Mallory could ask for a certificate on Bob's public key with her name in it:\\
    then whatever Bob signs would look as if Mallory had signed it.\end{center}
  \keymsg{A CSR proves the key, not the name.}
  \say{Podpis CSR weryfikujemy kluczem z tego samego CSR. To dowód posiadania klucza: nikt nie dostanie certyfikatu na cudzy klucz publiczny, którego prywatnej połowy nie ma. Bez tego Mallory mogłaby zgłosić publiczny klucz Boba pod swoją nazwą; podpisy Boba (na przykład pod umową albo w kliencie TLS) wyglądałyby wtedy na podpisy Mallory. Ale nazwę w CSR może wpisać każdy.}
\end{frame}
